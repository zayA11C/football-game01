import random class FarmGame: def __init__(self): self.or_ = 10 self.ble = 0 self.energie = 3 self.champs = 1 self.poules = 0 self.niveau = 1 self.jour = 1 # ---------- UTIL ---------- def status(self): print("\n--- STATUS ---") print(f"Jour: {self.jour}") print(f"Or: {self.or_}") print(f"Blé: {self.ble}") print(f"Énergie: {self.energie}") print(f"Champs: {self.champs}") print(f"Poules: {self.poules}") print("----------------\n") def event(self): events = [ ("🌧️ Pluie abondante", 1.5), ("🐀 Rats affamés", 0.9), ("🧑‍🌾 Marchand généreux", 1.2), ("⚡ Tempête", 0.8), ("☀️ Journée normale", 1.0), ] ev = random.choice(events) print(f"ÉVÉNEMENT : {ev[0]}") return ev[1] # ---------- ACTIONS ---------- def planter(self): gain = self.champs * 2 self.ble += gain print(f"🌱 Tu plantes et gagnes {gain} blé.") def recolter(self, bonus): gain = int(self.champs * 2 * bonus) self.ble += gain print(f"🌾 Tu récoltes {gain} blé.") def vendre(self): gain = self.ble // 2 self.or_ += gain print(f"🏪 Tu vends pour {gain} or.") self.ble = 0 def elever(self): self.poules += 1 print("🐔 Tu ajoutes une poule (+production future).") def ameliorer(self): if self.or_ >= 20: self.or_ -= 20 self.champs += 1 print("🔧 Amélioration: +1 champ") else: print("❌ Pas assez d'or (20 requis)") def repos(self): self.energie = 3 print("😴 Tu te reposes et récupères ton énergie.") # ---------- TOUR ---------- def jouer_tour(self): print(f"\n===== JOUR {self.jour} =====") bonus = self.event() actions = 3 self.energie = 3 while actions > 0: self.status() print("Actions possibles: planter / recolter / vendre / elever / ameliorer / repos") choix = input("Choisis une action: ").lower() if choix == "planter": self.planter() elif choix == "recolter": self.recolter(bonus) elif choix == "vendre": self.vendre() elif choix == "elever": self.elever() elif choix == "ameliorer": self.ameliorer() elif choix == "repos": self.repos() break else: print("Action inconnue.") continue actions -= 1 # production poules self.ble += self.poules * 2 print(f"🐔 Tes poules produisent {self.poules * 2} blé.") self.jour += 1 def jouer(self): while self.or_ < 10000: self.jouer_tour() if self.or_ >= 10000: print("\n🏆 TU AS GAGNÉ ! FERME LÉGENDAIRE !") break # ---------- LANCEMENT ---------- game = FarmGame() game.jouer()
